to get the blockers for D001, together with the evidence from the transcript:


PREFIX crm: <https://example.org/sales-kg/>
PREFIX res: <https://example.org/sales-kg/resource/>

SELECT ?blocker ?name ?speaker ?evidenceText
WHERE {
    res:Deal_D001 crm:hasBlocker ?blocker .

    OPTIONAL { ?blocker crm:name ?name }

    OPTIONAL {
        ?blocker crm:supportedBy ?evidence .
        ?evidence crm:speakerName ?speaker ;
                  crm:evidenceText ?evidenceText .
    }
}


To find the decision maker for D001:

PREFIX crm: <https://example.org/sales-kg/>
PREFIX res: <https://example.org/sales-kg/resource/>

SELECT ?person ?name
WHERE {
    ?person crm:decisionMakerFor res:Deal_D001 ;
            crm:name ?name .
}

A useful historical query is: which won deals involve the same product as D002?
PREFIX crm: <https://example.org/sales-kg/>
PREFIX res: <https://example.org/sales-kg/resource/>

SELECT ?wonDeal ?name ?value
WHERE {
    res:Deal_D002 crm:involvesProduct ?product .

    ?wonDeal
        a crm:Deal ;
        crm:involvesProduct ?product ;
        crm:hasOutcome crm:Won ;
        crm:name ?name ;
        crm:dealValue ?value .
}

And for expansion opportunities, D002 already has a good example:
PREFIX crm: <https://example.org/sales-kg/>
PREFIX res: <https://example.org/sales-kg/resource/>

SELECT ?opportunity ?name ?product ?evidenceText
WHERE {
    res:Deal_D002 crm:hasExpansionOpportunity ?opportunity .

    OPTIONAL { ?opportunity crm:name ?name }
    OPTIONAL { ?opportunity crm:expansionInvolvesProduct ?product }

    OPTIONAL {
        ?opportunity crm:supportedBy ?evidence .
        ?evidence crm:evidenceText ?evidenceText .
    }
}
